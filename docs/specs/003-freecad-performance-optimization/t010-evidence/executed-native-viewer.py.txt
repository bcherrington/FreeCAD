def field(obj):
 # The Pivy raw-byte overload decodes image bytes as UTF-8; use its ASCII field serialization.
 words=str(texture(obj).image.get()).split();width,height,components=map(int,words[:3]);dims=[width,height]
 if width==0 or height==0:return {'dimensions':dims,'components':components,'sha256':None}
 assert 0<width<=32767 and 0<height<=32767 and 1<=components<=4
 assert len(words[3:])==width*height
 raw=b''.join(int(word,0).to_bytes(components,'big') for word in words[3:])
 return {'dimensions':dims,'components':components,'sha256':fp(raw)}

   name=str(f).replace(str(root),'${REPO_ROOT}')
