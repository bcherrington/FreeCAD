#!/bin/sh
metacity --sm-disable > build/spec003-t010/metacity.log 2>&1 &
t010_wm_pid=$!
trap 'kill "$t010_wm_pid" 2>/dev/null || true' EXIT INT TERM
t010_attempt=0
while [ "$t010_attempt" -lt 30 ]; do
    if xprop -root _NET_SUPPORTING_WM_CHECK 2>/dev/null | rg -q 'window id'; then break; fi
    t010_attempt=$((t010_attempt + 1))
    sleep 0.1
done
"$@"
