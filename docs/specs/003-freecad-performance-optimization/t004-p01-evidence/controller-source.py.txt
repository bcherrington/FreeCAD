from pathlib import Path
import argparse,hashlib,importlib.util,json,os,subprocess,time
ROOT=Path.cwd();ART=ROOT/'build/spec003-t004-p01';SPEC=ROOT/'docs/specs/003-freecad-performance-optimization'
def require(ok,message):
 if not ok:raise ValueError(message)
def digest(path):
 with Path(path).open('rb') as stream:return hashlib.file_digest(stream,'sha256').hexdigest()
def write(path,data):
 temporary=path.with_suffix(path.suffix+'.tmp');temporary.write_text(json.dumps(data,indent=2)+'\n');temporary.replace(path)
def module(name,path):
 spec=importlib.util.spec_from_file_location(name,path);mod=importlib.util.module_from_spec(spec);spec.loader.exec_module(mod);return mod

def audit(freeze,phase):
 checked=0
 identity=json.loads((ART/'environment-identity.json').read_text());keys=list(identity['installed_packages'][0]);installed=[]
 for path in (ROOT/'.pixi/envs/default/conda-meta').glob('*.json'):
  package=json.loads(path.read_text());installed.append({key:package.get(key) for key in keys})
 installed.sort(key=lambda package:package['name']);require(installed==identity['installed_packages'],'Installed package inventory changed')
 for path,sha in freeze['files'].items():require(digest(path)==sha,'Frozen input changed');checked+=1
 for arm in ['baseline','change']:
  physical=ROOT/'build/spec003-t004-worktrees'/arm;build=physical/'build/relWithDebInfo';receipt=json.loads((ART/(arm+'-build-receipt.json')).read_text())
  require(subprocess.check_output(['git','-C',str(physical),'rev-parse','HEAD']).decode().strip()==receipt['commit'],'Wrong arm revision');require(not subprocess.check_output(['git','-C',str(physical),'status','--porcelain','--untracked-files=all']),'Dirty arm')
  for rel,sha in json.loads(Path(receipt['source_manifest']).read_text()).items():require(digest(physical/rel)==sha,'Arm source changed');checked+=1
  output_manifest=json.loads(Path(receipt['output_manifest']).read_text());current_outputs=set()
  for directory in ['bin','lib','Mod','Ext','share','data']:
   current_outputs.update(path.relative_to(build).as_posix() for path in (build/directory).rglob('*') if path.is_file())
  current_outputs.update(['CMakeCache.txt','compile_commands.json','tests/App_tests_run','tests/Base_tests_run','src/App/CMakeFiles/FreeCADApp.dir/PropertyFile.cpp.o','tests/src/App/CMakeFiles/App_tests_run.dir/PropertyFile.cpp.o'])
  require(current_outputs==set(output_manifest),'Runtime output inventory changed')
  for rel,data in output_manifest.items():require(digest(build/rel)==data['sha256'],'Build output changed');checked+=1
  for dep in receipt['dependencies']:
   child=physical/dep['path'];require(subprocess.check_output(['git','-C',str(child),'rev-parse','HEAD']).decode().strip()==dep['commit'],'Dependency revision changed');require(not subprocess.check_output(['git','-C',str(child),'status','--porcelain','--untracked-files=all']),'Dirty dependency')
 for entry in freeze['executions']:
  execution=json.loads(Path(entry['path']).read_text());require(digest(execution['fixture_path'])==execution['fixture_sha256'],'Fixture changed');checked+=1
 record={'schema':'freecad-spec003-p01-audit/v1','phase':phase,'passed':True,'checked_hashes':checked,'installed_packages_checked':len(installed),'unix_seconds':time.time()};write(ART/(phase+'-audit.json'),record);return record

def main():
 parser=argparse.ArgumentParser();parser.add_argument('--freeze-sha256',required=True);parser.add_argument('--audit-only',action='store_true');args=parser.parse_args();frozen=ART/'campaign-freeze.json';require(digest(frozen)==args.freeze_sha256,'Wrong campaign freeze');freeze=json.loads(frozen.read_text());require(digest(Path(__file__))==freeze['controller_sha256'],'Controller changed');require(os.environ.get('PYTHONOPTIMIZE')=='0','Unqualified namespace checker mode')
 require([entry['fixture_id'] for entry in freeze['executions']]==['gridfinity-drawer-v1.3','BIMExample','ArchDetail','EngineBlock','PartDesignExample'],'Wrong fixed cell inventory')
 status_path=ART/'collection-status.json'
 if args.audit_only:print(json.dumps(audit(freeze,'pre-collection')),flush=True);return
 require(not status_path.exists(),'Campaign already attempted')
 for i in range(5):require(not (ART/('observations-'+str(i)+'.jsonl')).exists(),'Preserve existing observations')
 audit(freeze,'before-collection')
 descriptor=os.open(status_path,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600);os.close(descriptor)
 status={'schema':'freecad-spec003-p01-collection/v1','freeze_sha256':args.freeze_sha256,'state':'running','started_unix_seconds':time.time(),'cells':[]};write(status_path,status)
 runner=module('p01_runner',SPEC/'benchmark-runner.py');diagnostic=module('p01_failure_collector',SPEC/'benchmark-image-collect.py');plan=json.loads((SPEC/'benchmark-plan.json').read_text())
 try:
  for i,entry in enumerate(freeze['executions']):
   require(digest(entry['path'])==entry['sha256'],'Execution changed');directory=ART/('failure-diagnostics-'+str(i));directory.mkdir(mode=0o700)
   identity='spec003-p01-'+args.freeze_sha256+'-cell-'+str(i)
   runner.collect=lambda command,environment,limits,timeout:diagnostic.collect(command,environment,limits,timeout,directory,identity)
   print('P01 cell',i,'started',flush=True);runner.run(plan,Path(entry['path']),ART/('observations-'+str(i)+'.jsonl'));status['cells'].append({'index':i,'fixture_id':entry['fixture_id'],'collection_complete':True});write(status_path,status);print('P01 cell',i,'complete',flush=True)
  audit(freeze,'after-collection');status['state']='complete';status['finished_unix_seconds']=time.time();write(status_path,status)
 except Exception as exc:
  status['state']='failed';status['error_type']=type(exc).__name__;status['failed_cell']=i;write(status_path,status);print('P01 collection stopped:',type(exc).__name__,flush=True);raise SystemExit(1)
if __name__=='__main__':main()
