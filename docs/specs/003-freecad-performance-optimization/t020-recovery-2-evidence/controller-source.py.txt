def require(condition, message):
    if not condition:
        raise ValueError(message)

from pathlib import Path
import argparse, hashlib, importlib.util, json, os, sys, time, traceback

def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value
p = argparse.ArgumentParser()
p.add_argument('freeze', type=Path)
p.add_argument('--sha256', required=True)
a = p.parse_args()
require(digest(a.freeze) == a.sha256, 'Freeze index changed')
freeze = json.loads(a.freeze.read_text())
require(digest(__file__) == freeze['controller']['sha256'], 'Controller changed')
art = Path(freeze['artifact_directory'])
pkg = Path(freeze['package_directory'])
status = art / 'collection-status.json'
require(not status.exists(), 'Never repeat a campaign')
rows = []

def verify():
    for item in freeze['bindings'].values():
        require(digest(item['path']) == item['sha256'], 'Frozen input changed')
verify()
helper = module('recovery_failure_collector', pkg / 'benchmark-image-collect.py')
for cell in ['hidden', 'visible', 'part', 'vrml', 'command-private', 'command-part', 'command-vrml']:
    entry = freeze['cells'][cell]
    require(not Path(entry['observations']).exists(), "Freeze guard failed: not Path(entry['observations']).exists()")
    require(not Path(entry['diagnostics']).exists(), "Freeze guard failed: not Path(entry['diagnostics']).exists()")
    require(digest(entry['execution']) == entry['execution_sha256'], "Freeze guard failed: digest(entry['execution']) == entry['execution_sha256']")
descriptor = os.open(status, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
with os.fdopen(descriptor, 'w') as stream:
    stream.write('[]\n')
for cell in ['hidden', 'visible', 'part', 'vrml', 'command-private', 'command-part', 'command-vrml']:
    verify()
    entry = freeze['cells'][cell]
    execution = Path(entry['execution'])
    record = json.loads(execution.read_text())
    require(digest(execution) == entry['execution_sha256'], "Freeze guard failed: digest(execution) == entry['execution_sha256']")
    out = Path(entry['observations'])
    diagnostics = Path(entry['diagnostics'])
    require(record['failure_collector_sha256'] == digest(pkg / 'benchmark-image-collect.py'), "Freeze guard failed: record['failure_collector_sha256'] == digest(pkg / 'benchmark-image-collect.py')")
    require(record['failure_diagnostics_dir'] == str(diagnostics), "Freeze guard failed: record['failure_diagnostics_dir'] == str(diagnostics)")
    identity = {'schema': 'freecad-spec003-diagnostic-run/v1', 'campaign': freeze['campaign'], 'execution_sha256': digest(execution), 'failure_collector_sha256': digest(pkg / 'benchmark-image-collect.py'), 'original_collector_sha256': digest(pkg / 'benchmark-runner.py'), 'context_sha256': record['context_sha256'], 'qualification_receipt_sha256': record['qualification_receipt_sha256']}
    helper.prepare_destination(diagnostics, identity)
    log = art / (cell + '-collection.log')
    started = time.time()
    exit_code = 0
    print('Starting fixed cell', cell, flush=True)
    with log.open('w') as stream:
        previous_stdout, previous_stderr = (sys.stdout, sys.stderr)
        sys.stdout = sys.stderr = stream
        try:
            if cell.startswith('command-'):
                collector = module('recovery_command_' + cell.replace('-', '_'), pkg / 'benchmark-command-collect.py')
                collector.runner.collect = lambda command, environment, limits, timeout: helper.collect(command, environment, limits, timeout, diagnostics, identity)
                collector.collect(execution, out)
            else:
                runner = helper.runner
                runner.collect = lambda command, environment, limits, timeout: helper.collect(command, environment, limits, timeout, diagnostics, identity)
                plan = json.loads(runner.PLAN.read_text())
                runner.check(plan)
                runner.run(plan, execution, out)
        except Exception:
            exit_code = 1
            traceback.print_exc()
        finally:
            sys.stdout, sys.stderr = (previous_stdout, previous_stderr)
    rows.append({'cell': cell, 'exit': exit_code, 'started_unix_seconds': started, 'finished_unix_seconds': time.time(), 'observations_lines': len(out.read_text().splitlines()) if out.exists() else 0, 'log_sha256': digest(log), 'diagnostic_failures': len(list(diagnostics.glob('failure-*/failure.json')))})
    status.write_text(json.dumps(rows, indent=2) + '\n')
    print('Finished cell', cell, 'exit', exit_code, 'lines', rows[-1]['observations_lines'], flush=True)
print('All seven corrected-campaign cells attempted once', flush=True)
